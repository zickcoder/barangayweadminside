5. Infrastructure as Code (IaC) — Database & Security

SQL scripts that automatically create and protect the database. No manual setup needed.

🕸 Supabase = cloud database
🔐 RLS = Row Level Security
⚡ Realtime = auto live updates
📂 SQL = script that builds tables
📧 .env = hidden secret keys
📋 Table 1 — Incident Reports
-- Create the incidents table
CREATE TABLE incoming_incidents (
  id            UUID   PRIMARY KEY,
  incident_id   TEXT   UNIQUE,    -- INC-001
  incident_type TEXT,             -- FIRE, FLOOD...
  priority      TEXT,             -- LOW / HIGH
  location      TEXT,             -- where it happened
  description   TEXT,             -- what happened
  status        TEXT   DEFAULT 'Pending',
  created_at    TIMESTAMPTZ DEFAULT now()
);

-- 🔐 Lock table (only allowed apps can use)
ALTER TABLE incoming_incidents
  ENABLE ROW LEVEL SECURITY;

-- Admin portal can read all reports
CREATE POLICY "Allow Read"
  ON incoming_incidents FOR SELECT USING(true);

-- Subsystems can submit reports
CREATE POLICY "Allow Insert"
  ON incoming_incidents FOR INSERT WITH CHECK(true);
📢 Table 2 — Broadcast Alerts
-- Create the alerts table
CREATE TABLE alerts (
  id             UUID   PRIMARY KEY,
  title          TEXT,              -- headline
  message        TEXT,              -- full message
  priority       TEXT,              -- WARNING / EMERGENCY
  emergency_type TEXT,              -- FIRE, FLOOD...
  language       TEXT   DEFAULT 'English',
  operator       TEXT   DEFAULT 'Administrator',
  incident_id    TEXT,              -- links to report
  created_at     TIMESTAMPTZ DEFAULT now()
);

-- 🔐 Lock table
ALTER TABLE alerts
  ENABLE ROW LEVEL SECURITY;

-- Mobile app can read all alerts
CREATE POLICY "Public Read"
  ON alerts FOR SELECT USING(true);

-- Admin portal can send new alerts
CREATE POLICY "Admin Insert"
  ON alerts FOR INSERT WITH CHECK(true);
⚡ Realtime — Live Auto-Updates
-- ⚡ Watch these tables for changes
-- Makes the app update without refresh

ALTER PUBLICATION supabase_realtime
  ADD TABLE incoming_incidents;

ALTER PUBLICATION supabase_realtime
  ADD TABLE alerts;

-- Result:
-- New report → Admin sees instantly
-- Admin broadcasts → Residents get
