9. System Process Flowchart — Emergency Response Workflow

flowchart TD
    START(["🚨 Emergency Incident Occurs"]) --> REPORT["External Agency Creates Incident Report"]
    REPORT --> POST["HTTP POST to /api/incoming-incidents"]
    POST --> VALID{"Payload Valid?"}
    VALID -- No --> ERR["Return 400 Error to Agency"]
    VALID -- Yes --> SAVE["Save to DB\nStatus: Pending"]
    SAVE --> REALTIME["Admin Portal Receives\nRealtime Update"]
    REALTIME --> QUEUE["Incident Appears in Queue\nwith Pending Badge"]

    QUEUE --> VIEW["Admin Clicks View Button"]
    VIEW --> MARK_SEEN["Status Updated:\nPending → Seen"]
    MARK_SEEN --> MODAL["Incident Detail Dialog Opens"]

    MODAL --> PATH{"Select Broadcast Method"}

    PATH -- AI Route --> GEN["Click Generate Summarized AI Alert"]
    GEN --> GEMINI["Gemini AI Processes\nIncident Description"]
    GEMINI --> AI_RESULT["AI Returns Official\nAlert Text Summary"]
    AI_RESULT --> CONFIRM_AI{"Admin Satisfied?"}
    CONFIRM_AI -- No, Regenerate --> GEN
    CONFIRM_AI -- Yes --> USE_AI["Click Use This Message Now"]

    PATH -- Default Route --> USE_DEF["Click Use This Message Too\n(Original description)"]

    USE_AI & USE_DEF --> PREFILL["Broadcast Form Pre-filled\nwith Selected Message"]
    PREFILL --> CLICK_BC["Admin Clicks Broadcast Button"]
    CLICK_BC --> PREVIEW["Preview Modal Opens\n(title, priority, message)"]
    PREVIEW --> APPROVE{"Admin Confirms?"}
    APPROVE -- Cancel --> PREFILL
    APPROVE -- Confirm --> INSERT_ALERT["INSERT Alert into DB\nStatus → Broadcasted"]
    INSERT_ALERT --> WS["Supabase Realtime\nWebSocket Event Triggered"]
    WS --> DELIVER["All Resident Apps\nReceive Alert Notification"]
    DELIVER --> CLEAR["Incident Removed\nfrom Active Queue"]
    CLEAR --> END(["✅ Emergency Alert\nSuccessfully Delivered"])
